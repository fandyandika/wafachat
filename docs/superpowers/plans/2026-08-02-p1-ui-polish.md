function DeltaPill({ value, format = (delta) => number.format(delta) }: {
  value: number;
  format?: (value: number) => string;

  if (!value) return null;
  const positive = value > 0;
  const label = format(Math.abs(value));
  return <span aria-label={`${positive ? "Naik" : "Turun"} ${label} dibanding periode sebelumnya`}>{positive ? "↑" : "↓"} {label}</span>;
}
